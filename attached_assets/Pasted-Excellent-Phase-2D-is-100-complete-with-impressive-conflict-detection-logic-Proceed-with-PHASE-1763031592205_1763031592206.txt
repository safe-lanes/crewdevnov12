  if (filters.crewMemberId) {
    conditions.push(eq(restHoursCrewRecords.crewMemberId, filters.crewMemberId));

  if (filters.month) {
    conditions.push(eq(restHoursCrewRecords.month, filters.month));
  }
  
  if (conditions.length === 0) {
    return await this.db.select().from(restHoursCrewRecords);
  }
  
  return await this.db
    .select()
    .from(restHoursCrewRecords)
    .where(and(...conditions));
}
