**`scripts/pre-deploy-check.js`** (works on Windows, Mac, Linux):
```javascript
const { execSync } = require('child_process');
const fs = require('fs');
const path = require('path');

console.log('🔍 Running pre-deployment regression check...\n');

try {
  // Run all tests
  console.log('Running tests...');
  execSync('npm test', { stdio: 'inherit' });
  
  // Check for TRUE regressions only (not new features)
  const regressionsFile = path.join('test-results', 'REGRESSIONS.md');
  
  if (fs.existsSync(regressionsFile)) {
    console.log('\n⚠️  REGRESSIONS DETECTED!');
    console.log('==================================');
    console.log(fs.readFileSync(regressionsFile, 'utf8'));
    console.log('==================================\n');
    console.log('❌ DEPLOYMENT BLOCKED');
    console.log('Reason: Existing functionality is broken');
    console.log('Action: Fix regressions before deploying\n');
    process.exit(1);
  }
  
  // Inform about new features (but don't block)
  const newFeaturesFile = path.join('test-results', 'NEW-FEATURES.md');
  
  if (fs.existsSync(newFeaturesFile)) {
    console.log('\nℹ️  New Features Detected');
    console.log('==================================');
    console.log(fs.readFileSync(newFeaturesFile, 'utf8'));
    console.log('==================================\n');
    console.log('✅ New features don\'t block deployment');
    
    const content = fs.readFileSync(newFeaturesFile, 'utf8');
    if (content.includes('Status: ❌ FAILED')) {
      console.log('⚠️  Warning: Some new feature tests are failing');
      console.log('These new features may not work correctly yet');
      console.log('Consider fixing before deploying\n');
    }
  }
  
  // All good - no TRUE regressions
  console.log('\n==================================');
  console.log('✅ ALL CHECKS PASSED');
  console.log('✅ NO REGRESSIONS DETECTED');
  console.log('✅ SAFE TO DEPLOY');
  console.log('==================================\n');
  process.exit(0);
  
} catch (error) {
  console.error('\n❌ Pre-deployment check failed');
  console.error('Error:', error.message);
  process.exit(1);
}
```

**`scripts/safety-check.js`** (works on Windows, Mac, Linux):
```javascript
const { execSync, spawn } = require('child_process');
const path = require('path');

console.log('🔍 Running safety checks...\n');

let checksPass = true;

// Check 1: Critical files unchanged (if using git)
try {
  console.log('1. Checking if application code unchanged...');
  const gitStatus = execSync('git diff --name-only client/ server/ shared/', { encoding: 'utf8' });
  
  if (gitStatus.trim()) {
    console.log('❌ ERROR: Application code was modified!');
    console.log('Changed files:');
    console.log(gitStatus);
    checksPass = false;
  } else {
    console.log('✅ Application code unchanged');
  }
} catch (error) {
  console.log('⚠️  Git not available or not a git repo - skipping check');
}

// Check 2: package.json has dev script
console.log('\n2. Checking if dev script exists...');
const packageJson = require(path.join(process.cwd(), 'package.json'));

if (packageJson.scripts && packageJson.scripts.dev) {
  console.log('✅ Dev script exists');
} else {
  console.log('❌ ERROR: Dev script missing!');
  checksPass = false;
}

// Check 3: Try to start app (with timeout)
console.log('\n3. Checking if app starts...');
console.log('⏳ Starting app (will timeout after 10 seconds)...');

const startApp = spawn('npm', ['run', 'dev'], {
  shell: true,
  stdio: 'pipe'
});

let appStarted = false;

startApp.stdout.on('data', (data) => {
  const output = data.toString();
  // Check for common success messages
  if (output.includes('Local:') || output.includes('localhost') || output.includes('ready')) {
    appStarted = true;
  }
});

// Wait 10 seconds then check
setTimeout(() => {
  startApp.kill();
  
  if (appStarted) {
    console.log('✅ App starts successfully\n');
  } else {
    console.log('❌ ERROR: App failed to start\n');
    checksPass = false;
  }
  
  // Final result
  if (checksPass) {
    console.log('==================================');
    console.log('✅ ALL SAFETY CHECKS PASSED');
    console.log('App is safe to use!');
    console.log('You can now run: npm test');
    console.log('==================================\n');
    process.exit(0);
  } else {
    console.log('==================================');
    console.log('❌ SAFETY CHECKS FAILED');
    console.log('Review errors above');
    console.log('==================================\n');
    process.exit(1);
  }
}, 10000);
```
