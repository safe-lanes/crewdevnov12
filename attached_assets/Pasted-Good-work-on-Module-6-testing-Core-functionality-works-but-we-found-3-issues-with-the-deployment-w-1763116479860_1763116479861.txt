**ISSUE 1: Deploy Assignment Endpoint - Missing Fields**

Problem: POST /api/rotation/proposals/deploy expects fields not in assignment structure
- Expected: crewId, joiningDate, contractPeriod
- Provided: crewMemberId, fromDate, toDate

Solution: Update the deploy endpoint to map assignment fields correctly

In server/routes.ts, find the deploy assignment endpoint and fix the field mapping:
```typescript
// POST /api/rotation/proposals/deploy
app.post('/api/rotation/proposals/deploy', async (req, res) => {
  try {
    const { planId, assignmentIndex, deployedBy } = req.body;
    
    // Get rotation plan
    const plan = await storage.getRotationPlan(planId);
    if (!plan) {
      return res.status(404).json({ error: 'Rotation plan not found' });
    }
    
    // Parse assignments
    const assignments = typeof plan.assignments === 'string' 
      ? JSON.parse(plan.assignments) 
      : plan.assignments;
    
    const assignment = assignments[assignmentIndex];
    if (!assignment) {
      return res.status(400).json({ error: 'Invalid assignment index' });
    }
    
    // Check for conflicts BEFORE deploying
    const conflicts = await storage.checkAssignmentConflicts(
      assignment.crewMemberId,
      assignment.vesselId,
      assignment.fromDate,
      assignment.toDate
    );
    
    if (conflicts && conflicts.length > 0) {
      return res.status(409).json({ 
        error: 'Assignment conflict detected',
        conflicts: conflicts 
      });
    }
    
    // Create vessel planning record with field mapping
    const vesselPlanningData = {
      crewMemberId: assignment.crewMemberId,
      vesselId: assignment.vesselId,
      rank: assignment.rank,
      joiningDate: assignment.fromDate,  // Map fromDate → joiningDate
      reliefDue: assignment.toDate,      // Map toDate → reliefDue
      contractPeriod: calculateMonths(assignment.fromDate, assignment.toDate), // Calculate months
      status: 'Deployed',
      deployedBy: deployedBy,
      deployedAt: new Date().toISOString()
    };
    
    // Create vessel planning record
    const vesselPlanning = await storage.createVesselPlanning(vesselPlanningData);
    
    // Update assignment status in rotation plan
    assignments[assignmentIndex].status = 'Deployed';
    assignments[assignmentIndex].deployedBy = deployedBy;
    assignments[assignmentIndex].deployedAt = new Date().toISOString();
    
    await storage.updateRotationPlan(planId, {
      assignments: JSON.stringify(assignments)
    });
    
    res.json({ 
      success: true, 
      vesselPlanningId: vesselPlanning.id,
      message: 'Assignment deployed successfully' 
    });
    
  } catch (error) {
    console.error('Deploy assignment error:', error);
    res.status(400).json({ error: 'Failed to deploy assignment' });
  }
});

// Helper function to calculate contract period in months
function calculateMonths(fromDate: string, toDate: string): number {
  const from = new Date(fromDate);
  const to = new Date(toDate);
  const months = (to.getFullYear() - from.getFullYear()) * 12 + 
                 (to.getMonth() - from.getMonth());
  return Math.max(1, months); // At least 1 month
}
```

```typescript
// GET /api/rotation/proposals/conflicts
app.get('/api/rotation/proposals/conflicts', async (req, res) => {
  try {
    const { 
      crewMemberId,     // Accept crewMemberId (standard)
      crewId,           // Also accept crewId (alternative)
      vesselId, 
      startDate,        // Accept startDate (standard)
      endDate,          // Accept endDate (standard)
      joiningDate,      // Also accept joiningDate (alternative)
      contractPeriod    // Also accept contractPeriod (alternative)
    } = req.query;
    
    // Use crewMemberId OR crewId
    const crew = crewMemberId || crewId;
    
    // Calculate endDate if contractPeriod provided
    let end = endDate;
    if (!end && joiningDate && contractPeriod) {
      const joining = new Date(joiningDate as string);
      joining.setMonth(joining.getMonth() + parseInt(contractPeriod as string));
      end = joining.toISOString().split('T')[0];
    }
    
    const start = startDate || joiningDate;
    
    if (!crew || !vesselId || !start || !end) {
      return res.status(400).json({ 
        error: 'Required parameters: crewMemberId (or crewId), vesselId, startDate (or joiningDate), endDate (or contractPeriod)' 
      });
    }
    
    const conflicts = await storage.checkAssignmentConflicts(
      crew as string,
      vesselId as string,
      start as string,
      end as string
    );
    
    res.json(conflicts || []);
    
  } catch (error) {
    console.error('Check conflicts error:', error);
    res.status(500).json({ error: 'Failed to check conflicts' });
  }
});
```
