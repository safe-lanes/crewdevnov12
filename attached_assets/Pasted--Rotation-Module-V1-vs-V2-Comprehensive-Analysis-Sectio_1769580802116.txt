### 2.2 Root Causes of Breakage

#### 🔴 ROOT CAUSE #1: Crew Pool Data is RANDOM (Critical)

**Symptom:** Pool filter in rotation doesn't work correctly

**Evidence:**
- File: `server/routes.ts`
- Lines: 6974-6975

```typescript
// PROBLEM: Random pool generation instead of real data
const pools = ['Pool A', 'Pool B', 'Pool C'];
const pool = pools[Math.floor(Math.random() * pools.length)];
```

**Why it happens:**
- The `/api/crew-members/by-rank/:rank` endpoint generates RANDOM pool values
- It calls `storage.getCrewMembers()` which returns V1 `crew_members` table
- The `crewPool` field in `crew_members` table is never populated
- V2 has `crew_personal_details.crew_pool` but this endpoint doesn't query it

**Impact:**
- Pool filter appears to work but filters against fake data
- Crew transferred from recruitment have real pool in `crew_personal_details`
- But rotation module gets random pool instead

**How to fix:**
1. Query `crew_personal_details.crew_pool` for V2 crew
2. Query `crew_members.crew_pool` for legacy crew
3. Join data properly in the endpoint

**Symptom:** Crew transferred from recruitment don't appear with correct pool

- V2 Schema: `shared/v2/crew-pool/schema.ts:89`

- Rotation module still queries the old `crew_members` table

**How to fix:**
1. Create new endpoint that queries V2 tables with JOIN
2. Or update existing endpoint to query both V1 and V2 crew
3. Map `crew_personal_details.crew_pool` to response

async function getCrewPoolForMember(crewId: string): Promise<string | null> {
  // Try V2 first (crew_members_v2 -> crew_personal_details)
  const v2Crew = await db.select()
    .from(crewMembersV2)
    .where(eq(crewMembersV2.empNo, crewId))
    .limit(1);

  if (v2Crew.length > 0) {
    const personalDetails = await db.select()
      .from(crewPersonalDetails)
      .where(eq(crewPersonalDetails.crewUuid, v2Crew[0].crewUuid))
      .limit(1);

    if (personalDetails.length > 0) {
      return personalDetails[0].crewPool;
    }
  }

  // Fall back to V1 crew_members.crew_pool
  const v1Crew = await db.select()
    .from(crewMembers)
    .where(eq(crewMembers.id, crewId))
    .limit(1);

  return v1Crew.length > 0 ? v1Crew[0].crewPool : null;
}

// ============================================
// CREW FOR ROTATION MODULE

router.get("/crew/by-rank/:rank", async (req, res) => {
  try {
    const { rank } = req.params;

    // Query V2 crew with personal details JOIN
    const v2Crew = await db.select({
      crewUuid: crewMembersV2.crewUuid,
      empNo: crewMembersV2.empNo,
      firstName: crewMembersV2.firstName,
      middleName: crewMembersV2.middleName,
      familyName: crewMembersV2.familyName,
      presentRank: crewMembersV2.presentRank,
      nationality: crewMembersV2.nationalityUuid,
      status: crewMembersV2.status,
      crewPool: crewPersonalDetails.crewPool,
      availability: crewPersonalDetails.availability,
      nextAvailability: crewPersonalDetails.nextAvailability,
      manningAgent: crewPersonalDetails.manningAgent,
    })
    .from(crewMembersV2)
    .leftJoin(crewPersonalDetails, eq(crewMembersV2.crewUuid, crewPersonalDetails.crewUuid))
    .where(and(
      eq(crewMembersV2.presentRank, rank),
      eq(crewMembersV2.status, 'Active'),
      eq(crewMembersV2.isDeleted, false)
    ));

    // Format response for rotation module
    const formattedCrew = v2Crew.map(crew => ({
      id: crew.empNo,
      crewUuid: crew.crewUuid,
      name: `${crew.firstName} ${crew.middleName || ''} ${crew.familyName || ''}`.trim(),
      rank: crew.presentRank,
      pool: crew.crewPool || null,
      manningAgent: crew.manningAgent,
      nationality: crew.nationality,
      travelStatus: crew.availability || 'Available',
      nextAvailability: crew.nextAvailability,
      isV2: true, // Flag for V2 crew
    }));

    res.json({ success: true, data: formattedCrew });
  } catch (error) {
    console.error("Failed to fetch V2 crew by rank:", error);
    res.status(500).json({ success: false, error: "Failed to fetch crew" });
