## EXAMPLE PATTERN TO FOLLOW:

```typescript
import { describe, it, expect } from 'vitest';
import { insertRecruitmentCandidateSchema } from '@shared/schema';

describe('Recruitment Candidate Schema Validation', () => {
  describe('Valid Data', () => {
    it('should validate complete candidate data', () => {
      const validCandidate = {
        firstName: 'John',
        lastName: 'Doe',
        rankApplied: 'Chief Engineer',
        nationality: 'Indian',
        vesselType: 'Tanker',
        status: 'Screening',
        email: 'john@example.com',
        phone: '+91-9876543210'
      };

      const result = insertRecruitmentCandidateSchema.safeParse(validCandidate);
      expect(result.success).toBe(true);
      if (result.success) {
        expect(result.data.firstName).toBe('John');
      }
    });

    it('should accept optional fields as undefined', () => {
      const minimalCandidate = {
        firstName: 'Jane',
        lastName: 'Smith',
        rankApplied: 'AB',
        nationality: 'Filipino',
        status: 'Screening'
      };

      const result = insertRecruitmentCandidateSchema.safeParse(minimalCandidate);
      expect(result.success).toBe(true);
    });
  });

  describe('Invalid Data', () => {
    it('should reject missing firstName', () => {
      const invalid = {
        lastName: 'Doe',
        rankApplied: 'AB',
        nationality: 'Indian',
        status: 'Screening'
      };

      const result = insertRecruitmentCandidateSchema.safeParse(invalid);
      expect(result.success).toBe(false);
      if (!result.success) {
        expect(result.error.issues[0].path).toContain('firstName');
      }
    });

    it('should reject invalid email format', () => {
      const invalid = {
        firstName: 'John',
        lastName: 'Doe',
        rankApplied: 'AB',
        nationality: 'Indian',
        status: 'Screening',
        email: 'not-an-email'
      };

      const result = insertRecruitmentCandidateSchema.safeParse(invalid);
      expect(result.success).toBe(false);
    });

    it('should reject invalid status', () => {
      const invalid = {
        firstName: 'John',
        lastName: 'Doe',
        rankApplied: 'AB',
        nationality: 'Indian',
        status: 'InvalidStatus'
      };

      const result = insertRecruitmentCandidateSchema.safeParse(invalid);
      expect(result.success).toBe(false);
    });
  });

  describe('Business Logic', () => {
    it('should validate experience requirements for rank', () => {
      // Test business logic using validated data
      const candidate = {
        firstName: 'John',
        lastName: 'Doe',
        rankApplied: 'Chief Engineer',
        nationality: 'Indian',
        status: 'Screening',
        experience: 12
      };

      const result = insertRecruitmentCandidateSchema.safeParse(candidate);
      expect(result.success).toBe(true);

      // Then test business rule
      if (result.success) {
        const rankMinExperience: Record<string, number> = {
          'Chief Engineer': 10,
          'Third Officer': 3
        };
        expect(result.data.experience || 0).toBeGreaterThanOrEqual(
          rankMinExperience[result.data.rankApplied] || 0
        );
      }
    });
  });
});
```
