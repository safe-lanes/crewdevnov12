### 1. Test Setup File

Create `tests/setup.ts`:

```typescript
import { beforeAll, afterAll, afterEach } from 'vitest';
import '@testing-library/jest-dom';
import fs from 'fs';
import path from 'path';

// Create test results directories
beforeAll(() => {
  const dirs = ['test-results', 'test-results/screenshots', 'test-logs', 'coverage'];
  dirs.forEach(dir => {
    if (!fs.existsSync(dir)) {
      fs.mkdirSync(dir, { recursive: true });
    }
  });
  
  // Create test run log
  const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
  const logFile = path.join('test-logs', `test-run-${timestamp}.log`);
  global.testLogFile = logFile;
  fs.writeFileSync(logFile, `Test Run Started: ${new Date().toISOString()}\n`);
  fs.appendFileSync(logFile, '='.repeat(80) + '\n\n');
});

afterEach((context) => {
  // Log each test result
  if (global.testLogFile && context) {
    const status = context.task.result?.state || 'unknown';
    const duration = context.task.result?.duration || 0;
    const log = `[${status.toUpperCase()}] ${context.task.name} (${duration}ms)\n`;
    fs.appendFileSync(global.testLogFile, log);
    
    if (status === 'fail' && context.task.result?.errors) {
      context.task.result.errors.forEach(error => {
        fs.appendFileSync(global.testLogFile, `  Error: ${error.message}\n`);
        fs.appendFileSync(global.testLogFile, `  Stack: ${error.stack}\n\n`);
      });
    }
  }
});

afterAll(() => {
  if (global.testLogFile) {
    fs.appendFileSync(global.testLogFile, '\n' + '='.repeat(80) + '\n');
    fs.appendFileSync(global.testLogFile, `Test Run Completed: ${new Date().toISOString()}\n`);
  }
});
```

```typescript
  const hours = Array(48).fill('');
  restPeriods.forEach(([start, duration]) => {
    for (let i = start; i < start + (duration * 2); i++) {
      if (i < 48) hours[i] = '';

  });
  // Fill remaining with work
  return hours.map(h => h === '' ? 'w' : h);
```

```javascript
// tests/helpers/regression-detector.js
function analyzeTestResults(currentResults, baseline) {
  const analysis = {
    regressions: [],        // Previously passing, now failing
    newTests: [],           // Tests that didn't exist in baseline
    stillPassing: [],       // Tests still passing
    stillFailing: []        // Tests that were already failing
  };
  
  // Analyze current test results
  for (const test in currentResults) {
    const currentStatus = currentResults[test];
    const baselineStatus = baseline[test];
    
    if (!baselineStatus) {
      // Test didn't exist in baseline - NEW FEATURE
      analysis.newTests.push({
        test: test,
        status: currentStatus,
        note: 'New test - likely for new feature'
      });
    } else if (baselineStatus === 'passed' && currentStatus === 'failed') {
      // Was passing, now failing - TRUE REGRESSION
      analysis.regressions.push({
        test: test,
        status: 'REGRESSION',
        severity: 'HIGH',
        lastPassed: baseline.timestamp
      });
    } else if (baselineStatus === 'passed' && currentStatus === 'passed') {
      // Still passing - good
      analysis.stillPassing.push(test);
    } else if (baselineStatus === 'failed' && currentStatus === 'failed') {
      // Was already failing - not a new regression
      analysis.stillFailing.push({
        test: test,
        note: 'Known failure - was failing in baseline'
      });
    }
  }
  
  return analysis;
}

function generateRegressionReport(analysis) {
  let report = '# Test Analysis Report\n\n';
  
  // Only flag TRUE regressions
  if (analysis.regressions.length > 0) {
    report += '## ⚠️ REGRESSIONS DETECTED\n\n';
    report += 'These tests were PASSING but now FAIL:\n\n';
    analysis.regressions.forEach(r => {
      report += `### ${r.test}\n`;
      report += `- **Status:** REGRESSION\n`;
      report += `- **Last Passed:** ${r.lastPassed}\n`;
      report += `- **Action Required:** Review recent code changes\n\n`;
    });
    report += '❌ **DO NOT DEPLOY until regressions are fixed!**\n\n';
  }
  
  // Report new tests separately (NOT as regressions)
  if (analysis.newTests.length > 0) {
    report += '## ℹ️ New Tests Detected\n\n';
    report += 'These tests are new (likely for new features):\n\n';
    analysis.newTests.forEach(t => {
      const icon = t.status === 'passed' ? '✅' : '❌';
      report += `- ${icon} ${t.test} (${t.status})\n`;
    });
    report += '\n';
    report += '**Note:** New tests are NOT regressions. ';
    report += 'Review to ensure new features work correctly.\n\n';
  }
  
  // Summary
  report += '## Summary\n\n';
  report += `- Regressions: ${analysis.regressions.length}\n`;
  report += `- New Tests: ${analysis.newTests.length}\n`;
  report += `- Still Passing: ${analysis.stillPassing.length}\n`;
  report += `- Known Failures: ${analysis.stillFailing.length}\n\n`;
  
  return report;
}
```

**4. Smart Deployment Decision:**

```javascript
// tests/helpers/deployment-check.js
function canDeploy(analysis) {
  const decision = {
    allowed: true,
    reason: '',
    warnings: []
  };
  
  // Block deployment only for TRUE regressions
  if (analysis.regressions.length > 0) {
    decision.allowed = false;
    decision.reason = `${analysis.regressions.length} regression(s) detected - existing functionality broken`;
    return decision;
  }
  
  // Warn about new failing tests, but don't block
  const newFailures = analysis.newTests.filter(t => t.status === 'failed');
  if (newFailures.length > 0) {
    decision.warnings.push(
      `${newFailures.length} new test(s) failing - new features may be incomplete`
    );
  }
  
  // Warn about known failures  
  if (analysis.stillFailing.length > 0) {
    decision.warnings.push(
      `${analysis.stillFailing.length} known failing test(s) - pre-existing issues`
    );
  }
  
  return decision;
}
```
