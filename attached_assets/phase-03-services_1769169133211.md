### 1. `crewMembersService.ts`

```typescript
import { crewMembersRepository } from '../repositories';
import type { InsertCrewMemberV2, CrewMemberV2 } from '@shared/v2/crew-pool/types';

export const crewMembersService = {
  async getAll(filters?: { status?: string; isActive?: boolean; search?: string }) {
    return crewMembersRepository.findAll(filters);
  },

  async getByUuid(crewUuid: string): Promise<CrewMemberV2> {
    const crew = await crewMembersRepository.findByUuid(crewUuid);
    if (!crew) {
      throw new Error(`Crew member not found: ${crewUuid}`);
    }
    return crew;
  },

  async create(data: Omit<InsertCrewMemberV2, 'crewUuid'>): Promise<CrewMemberV2> {
    // Validate required fields
    if (!data.firstName || !data.familyName) {
      throw new Error('First name and family name are required');
    }
    if (!data.empNo) {
      throw new Error('Employee number is required');
    }
    
    return crewMembersRepository.create(data);
  },

  async update(crewUuid: string, data: Partial<InsertCrewMemberV2>): Promise<CrewMemberV2> {
    // Verify exists
    await this.getByUuid(crewUuid);
    
    const updated = await crewMembersRepository.update(crewUuid, data);
    if (!updated) {
      throw new Error(`Failed to update crew member: ${crewUuid}`);
    }
    return updated;
  },

  async archive(crewUuid: string): Promise<void> {
    // Verify exists
    await this.getByUuid(crewUuid);
    await crewMembersRepository.softDelete(crewUuid);
  },

  async uploadPhoto(crewUuid: string, photoPath: string): Promise<CrewMemberV2> {
    return this.update(crewUuid, { uploadedPhoto: photoPath });
  },
};
```

### 2. `crewAssignmentsService.ts`

```typescript
import { crewAssignmentsRepository } from '../repositories';
import { crewMembersService } from './crewMembersService';
import type { InsertCrewAssignment, CrewAssignment } from '@shared/v2/crew-pool/types';

export const crewAssignmentsService = {
  async getAll(crewUuid: string): Promise<CrewAssignment[]> {
    // Verify crew exists
    await crewMembersService.getByUuid(crewUuid);
    return crewAssignmentsRepository.findByCrewUuid(crewUuid);
  },

  async getCurrent(crewUuid: string): Promise<CrewAssignment | null> {
    await crewMembersService.getByUuid(crewUuid);
    const current = await crewAssignmentsRepository.findCurrent(crewUuid);
    return current || null;
  },

  async create(crewUuid: string, data: Omit<InsertCrewAssignment, 'assignUuid' | 'crewUuid'>): Promise<CrewAssignment> {
    await crewMembersService.getByUuid(crewUuid);
    
    // Validate dates
    if (data.signOnDate && data.signOffDate) {
      if (new Date(data.signOnDate) > new Date(data.signOffDate)) {
        throw new Error('Sign on date cannot be after sign off date');
      }
    }
    
    return crewAssignmentsRepository.create({ ...data, crewUuid });
  },

  async update(assignUuid: string, data: Partial<InsertCrewAssignment>): Promise<CrewAssignment> {
    const updated = await crewAssignmentsRepository.update(assignUuid, data);
    if (!updated) {
      throw new Error(`Assignment not found: ${assignUuid}`);
    }
    return updated;
  },

  async delete(assignUuid: string): Promise<void> {
    await crewAssignmentsRepository.delete(assignUuid);
  },
};
```

import { crewMembersService } from './crewMembersService';
import { crewAssignmentsService } from './crewAssignmentsService';
import { crewPersonalRepository, crewFamilyRepository, crewVesselTypesRepository } from '../repositories';

export const crewProfileService = {

      crew,

      personal,
      address,

      nextOfKin,
      vesselTypes,

      crewFamilyRepository.findChildrenByCrewUuid(crewUuid),
      crewFamilyRepository.findNokByCrewUuid(crewUuid),
      crewVesselTypesRepository.findByCrewUuid(crewUuid),
    ]);

    return {
      crew,
      currentAssignment,
      personal,
      address,
      family: {
        ...family,
        children,
        nextOfKin,
      },
      vesselTypes,
    };
  },
};

import { crewSeaServiceRepository } from '../repositories';
import { crewMembersService } from './crewMembersService';

export const crewSeaServiceService = {
  async getByType(crewUuid: string, serviceType: 'company' | 'external') {
    await crewMembersService.getByUuid(crewUuid);
    return crewSeaServiceRepository.findByCrewUuidAndType(crewUuid, serviceType);
  },

  async getAll(crewUuid: string) {
    await crewMembersService.getByUuid(crewUuid);
    return crewSeaServiceRepository.findByCrewUuidWithAttachments(crewUuid);
  },

  // Calculate total sea service duration
  async getTotalExperience(crewUuid: string) {
    const services = await this.getAll(crewUuid);
    
    let totalDays = 0;
    for (const service of services) {
      if (service.signOnDate && service.signOffDate) {
        const start = new Date(service.signOnDate);
        const end = new Date(service.signOffDate);
        totalDays += Math.ceil((end.getTime() - start.getTime()) / (1000 * 60 * 60 * 24));
      }
    }
    
    return {
      totalDays,
      totalMonths: Math.floor(totalDays / 30),
      totalYears: Math.floor(totalDays / 365),
    };
  },

  // ... other CRUD methods
};

## Key Patterns

### 1. Always Verify Crew Exists
```typescript
async getSomething(crewUuid: string) {
  await crewMembersService.getByUuid(crewUuid); // Throws if not found
  return repository.findByCrewUuid(crewUuid);
}
```

### 2. Throw Errors for Not Found
```typescript
const result = await repository.findByUuid(uuid);
if (!result) {
  throw new Error(`Resource not found: ${uuid}`);
}
return result;
```

### 3. Use Promise.all for Parallel Fetches
```typescript
const [a, b, c] = await Promise.all([
  fetchA(),
  fetchB(),
  fetchC(),
]);
```
