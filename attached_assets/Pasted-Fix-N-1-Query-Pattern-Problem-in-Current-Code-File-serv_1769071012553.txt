export class VesselTypesAppliedRepository {
  // ... existing methods ...

  /**
   * Batch load vessel types for multiple candidates in a single query
   * Replaces N+1 pattern with single query
   */
  async findByCandidateUuids(recCanUuids: string[]): Promise<Map<string, CandVesselTypeApplied[]>> {
    if (recCanUuids.length === 0) {
      return new Map();
    }

    const db = getDb();
    const vesselTypes = await db
      .select()
      .from(candVesselTypesApplied)
      .where(
        and(
          sql`${candVesselTypesApplied.recCanUuid} = ANY(${recCanUuids})`,
          eq(candVesselTypesApplied.isDeleted, false)
        )
      )
      .orderBy(candVesselTypesApplied.sortOrder);

    // Group by candidate UUID
    const vesselTypeMap = new Map<string, CandVesselTypeApplied[]>();
    for (const vt of vesselTypes) {
      if (!vesselTypeMap.has(vt.recCanUuid)) {
        vesselTypeMap.set(vt.recCanUuid, []);
      }
      vesselTypeMap.get(vt.recCanUuid)!.push(vt);
    }

    return vesselTypeMap;
  }
}

async getAllCandidates(): Promise<CandidateListItem[]> {
  const candidates = await candidateRepository.findAll();

  // Single batch query instead of N queries
  const recCanUuids = candidates.map(c => c.recCanUuid);
  const vesselTypeMap = await vesselTypesAppliedRepository.findByCandidateUuids(recCanUuids);

  return candidates.map(candidate => {
    const vesselTypes = vesselTypeMap.get(candidate.recCanUuid) || [];
    const primaryVesselType = vesselTypes.length > 0 ? vesselTypes[0].vesselTypeUuid || '' : '';

    return {
      ...candidate,
      nationality: candidate.nationalityUuid || '',
      vesselType: primaryVesselType,
    };
  });
}
