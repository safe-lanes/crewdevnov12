#### 2.2: Create/Update Soft Delete Endpoint

**Add PATCH endpoint for soft delete:**

```typescript
// Soft delete a recruitment candidate (mark as deleted)
app.patch("/api/recruitment-candidates/:id/delete", async (req, res) => {
  try {
    const { id } = req.params;
    
    log(`Soft deleting recruitment candidate: ${id}`, 'api');
    
    // Update the record to set isDelete = 1
    const updated = await storage.updateRecruitmentCandidate(id, {
      isDelete: 1,  // or true for BOOLEAN
      updatedAt: new Date()
    });
    
    if (!updated) {
      return res.status(404).json({ 
        error: "Candidate not found",
        message: `Recruitment candidate with ID ${id} does not exist`
      });
    }
    
    log(`Successfully soft deleted candidate: ${id}`, 'api');
    
    res.json({ 
      success: true,
      message: "Recruitment candidate deleted successfully",
      id: id
    });
    
  } catch (error) {
    console.error('Error soft deleting recruitment candidate:', error);
    res.status(500).json({ 
      error: "Failed to delete recruitment candidate",
      message: error.message 
    });
  }
});

// Alternative: Use standard PATCH with isDelete in body
app.patch("/api/recruitment-candidates/:id", async (req, res) => {
  try {
    const { id } = req.params;
    const result = updateRecruitmentCandidateSchema.safeParse(req.body);
    
    if (!result.success) {
      return res.status(400).json({ 
        error: "Invalid data", 
        details: result.error.issues 
      });
    }
    
    const updated = await storage.updateRecruitmentCandidate(id, result.data);
    
    if (!updated) {
      return res.status(404).json({ error: "Candidate not found" });
    }
    
    // If isDelete was set to 1, log it
    if (result.data.isDelete === 1 || result.data.isDelete === true) {
      log(`Candidate ${id} soft deleted`, 'api');
    }
    
    res.json(updated);
    
  } catch (error) {
    console.error('Error updating recruitment candidate:', error);
    res.status(500).json({ error: "Failed to update recruitment candidate" });
  }
});
```

**Choose one approach:**
- **Option A:** Dedicated `/delete` endpoint (cleaner, more explicit)
- **Option B:** Use standard PATCH with `isDelete` in body (more flexible)

**Fixed delete handler (CORRECT):**
```typescript
const handleDeleteClick = async (params: any) => {
  const candidateId = params.data.id;
  const candidateName = `${params.data.firstName} ${params.data.lastName}` || params.data.fileNo;
  
  // Show confirmation dialog
  const confirmed = window.confirm(
    `Are you sure you want to delete "${candidateName}"?\n\n` +
    `This will mark the candidate as deleted and remove them from the list.\n` +
    `File No: ${params.data.fileNo}`
  );
  
  if (!confirmed) {
    return; // User cancelled
  }
  
  try {
    // Call soft delete API
    const response = await fetch(
      `/api/recruitment-candidates/${candidateId}/delete`,
      {
        method: 'PATCH',
        headers: {
          'Content-Type': 'application/json',
        },
      }
    );
    
    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.message || 'Failed to delete candidate');
    }
    
    const result = await response.json();
    
    // Show success message
    toast.success(`Candidate deleted successfully`);
    // or alert(`Candidate deleted successfully`);
    
    // Refresh the table to remove the deleted record
    await refetch(); // If using React Query
    // or window.location.reload(); // Simple but not ideal
    // or setRefreshKey(prev => prev + 1); // If using refresh key pattern
    
  } catch (error) {
    console.error('Error deleting candidate:', error);
    toast.error(`Failed to delete candidate: ${error.message}`);
    // or alert(`Failed to delete candidate: ${error.message}`);
  }
};

// Alternative using standard PATCH with isDelete in body:
const handleDeleteClick = async (params: any) => {
  const candidateId = params.data.id;
  const candidateName = `${params.data.firstName} ${params.data.lastName}` || params.data.fileNo;
  
  const confirmed = window.confirm(
    `Are you sure you want to delete "${candidateName}"?`
  );
  
  if (!confirmed) return;
  
  try {
    const response = await fetch(
      `/api/recruitment-candidates/${candidateId}`,
      {
        method: 'PATCH',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          isDelete: 1  // Mark as deleted
        })
      }
    );
    
    if (!response.ok) {
      throw new Error('Failed to delete candidate');
    }
    
    toast.success('Candidate deleted successfully');
    refetch(); // Refresh table
    
  } catch (error) {
    console.error('Error deleting candidate:', error);
    toast.error(`Failed to delete: ${error.message}`);
  }
};
```

## 🚨 IMPORTANT NOTES

### 1. Column Naming
- Database column: `is_delete` (snake_case)
- TypeScript/API: `isDelete` (camelCase)
- Ensure proper mapping between formats

### 2. Soft Delete vs Hard Delete
- ✅ Soft delete: Set `is_delete = 1` (record kept in database)
- ❌ Hard delete: `DELETE FROM table` (record removed forever)
- **Use soft delete only** - allows recovery if needed

### 3. Filtering Considerations
- GET endpoint MUST filter `is_delete = 1` records
- Consider performance with large datasets
- Index on `is_delete` column helps performance
