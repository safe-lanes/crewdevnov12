async deployAssignment(planId: number, assignmentIndex: number, deployedBy: string): Promise<boolean> {
  try {
    // Get current plan
    const plans = await this.db
      .select()
      .from(rotationPlans)
      .where(eq(rotationPlans.id, planId));
    
    if (!plans[0]) return false;
    
    // Parse and update assignments
    const assignments = JSON.parse(plans[0].assignments || '[]');
    if (!assignments[assignmentIndex]) return false;
    
    assignments[assignmentIndex] = {
      ...assignments[assignmentIndex],
      status: 'Deployed',
      deployedBy: deployedBy,
      deployedAt: new Date().toISOString()
    };
    
    // Save updated assignments
    await this.db
      .update(rotationPlans)
      .set({ 
        assignments: JSON.stringify(assignments),
        lastEdited: new Date()
      })
      .where(eq(rotationPlans.id, planId));
    
    return true;
  } catch (error) {
    console.error('Error deploying assignment:', error);
    return false;
  }
}
