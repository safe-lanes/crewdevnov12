import { insertRecruitmentCandidateSchema } from '@shared/schema';

describe('Recruitment Candidate Validation', () => {
  it('should validate complete recruitment candidate data', () => {
    const validCandidate = {
      firstName: 'John',
      lastName: 'Doe',
      rankApplied: 'Chief Engineer',
      nationality: 'Indian',
      vesselType: 'Tanker',
      status: 'Screening',
      email: 'john.doe@example.com',
      phone: '+91-9876543210',
      dateOfBirth: '1990-05-15',
      experience: 8
    };
    
    const result = insertRecruitmentCandidateSchema.safeParse(validCandidate);
    expect(result.success).toBe(true);
  });

  it('should reject candidate without required fields', () => {
    const invalidData = { firstName: 'John' };
    const result = insertRecruitmentCandidateSchema.safeParse(invalidData);
    expect(result.success).toBe(false);
  });

  it('should validate email format', () => {
    const validEmail = 'john@example.com';
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    expect(emailRegex.test(validEmail)).toBe(true);
  });

  it('should validate status transitions', () => {
    const validStatuses = ['Screening', 'Interview', 'Offered', 'Joined', 'Rejected'];
    const currentStatus = 'Screening';
    const nextStatus = 'Interview';
    
    const currentIndex = validStatuses.indexOf(currentStatus);
    const nextIndex = validStatuses.indexOf(nextStatus);
    
    // Can only move forward in workflow (except Rejected)
    expect(nextIndex >= currentIndex || nextStatus === 'Rejected').toBe(true);
  });

    const seaServiceMonths = 48;
    const experienceYears = seaServiceMonths / 12;
    expect(experienceYears).toBe(4);

import { describe, it, expect, beforeEach } from 'vitest';

describe('Recruitment API', () => {
  let testCandidateId: number;

  describe('POST /api/recruitment-candidates', () => {
    it('should create new recruitment candidate', async () => {
      const candidate = {
        firstName: 'John',
        lastName: 'Doe',
        rankApplied: 'Chief Engineer',
        nationality: 'Indian',
        vesselType: 'Tanker',
        status: 'Screening',
        email: 'test@example.com',
        phone: '+91-9876543210'
      };

      const response = await fetch('/api/recruitment-candidates', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(candidate)
      });

      expect(response.status).toBe(201);
      const data = await response.json();
      expect(data.id).toBeDefined();
      testCandidateId = data.id;
    });

    it('should validate required fields', async () => {
      const response = await fetch('/api/recruitment-candidates', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ firstName: 'Incomplete' })
      });

      expect(response.status).toBe(400);
    });

    it('should prevent duplicate email addresses', async () => {
      const candidate = {
        firstName: 'Jane',
        lastName: 'Smith',
        email: 'duplicate@example.com',
        rankApplied: 'AB',
        nationality: 'Filipino',
        status: 'Screening'
      };

      // Create first
      await fetch('/api/recruitment-candidates', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(candidate)
      });

      // Try to create duplicate
      const response = await fetch('/api/recruitment-candidates', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(candidate)
      });

      expect(response.status).toBe(400);
    });
  });

  describe('GET /api/recruitment-candidates', () => {
    it('should list all recruitment candidates', async () => {
      const response = await fetch('/api/recruitment-candidates');
      expect(response.status).toBe(200);
      
      const data = await response.json();
      expect(Array.isArray(data)).toBe(true);
    });

    it('should filter candidates by status', async () => {
      const response = await fetch('/api/recruitment-candidates?status=Screening');
      const data = await response.json();
      
      data.forEach((candidate: any) => {
        expect(candidate.status).toBe('Screening');
      });
    });

    it('should search candidates by name', async () => {
      const response = await fetch('/api/recruitment-candidates?search=John');
      const data = await response.json();
      
      data.forEach((candidate: any) => {
        const fullName = `${candidate.firstName} ${candidate.lastName}`.toLowerCase();
        expect(fullName).toContain('john');
      });
    });

    it('should filter by rank applied', async () => {
      const response = await fetch('/api/recruitment-candidates?rank=Chief%20Engineer');
      const data = await response.json();
      
      data.forEach((candidate: any) => {
        expect(candidate.rankApplied).toBe('Chief Engineer');
      });
    });
  });

  describe('PUT /api/recruitment-candidates/:id', () => {
    it('should update candidate status', async () => {
      const response = await fetch(`/api/recruitment-candidates/${testCandidateId}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ status: 'Interview' })
      });

      expect(response.status).toBe(200);
      const updated = await response.json();
      expect(updated.status).toBe('Interview');
    });

    it('should update candidate details', async () => {
      const updates = {
        phone: '+91-9999999999',
        email: 'updated@example.com'
      };

      const response = await fetch(`/api/recruitment-candidates/${testCandidateId}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(updates)
      });

      expect(response.status).toBe(200);
    });

    it('should reject invalid status update', async () => {
      const response = await fetch(`/api/recruitment-candidates/${testCandidateId}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ status: 'InvalidStatus' })
      });

      expect(response.status).toBe(400);
    });
  });

  describe('POST /api/recruitment-candidates/:id/transfer-to-crew', () => {
    it('should transfer candidate to crew pool', async () => {
      // First update status to 'Offered'
      await fetch(`/api/recruitment-candidates/${testCandidateId}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ status: 'Offered' })
      });

      // Then transfer
      const response = await fetch(
        `/api/recruitment-candidates/${testCandidateId}/transfer-to-crew`,
        { method: 'POST' }
      );

      expect(response.status).toBe(200);
      const result = await response.json();
      expect(result.crewMemberId).toBeDefined();
    });

    it('should reject transfer if status is not Offered', async () => {
      const response = await fetch(
        `/api/recruitment-candidates/999/transfer-to-crew`,
        { method: 'POST' }
      );

      expect(response.status).toBe(400);
    });
  });

  describe('DELETE /api/recruitment-candidates/:id', () => {
    it('should delete candidate', async () => {
      const response = await fetch(`/api/recruitment-candidates/${testCandidateId}`, {
        method: 'DELETE'
      });

      expect(response.status).toBe(200);
    });

    it('should return 404 for non-existent candidate', async () => {
      const response = await fetch('/api/recruitment-candidates/99999', {
        method: 'DELETE'
      });

      expect(response.status).toBe(404);
    });
  });
});
